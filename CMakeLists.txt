cmake_minimum_required(VERSION 3.16)
project(knulli_random_game_picker VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2 SDL2_image SDL2_ttf)
find_package(pugixml REQUIRED)
find_package(nlohmann_json REQUIRED)
add_library(picker-library src/library.cpp)
target_include_directories(picker-library PUBLIC src)
target_link_libraries(picker-library PUBLIC pugixml::static nlohmann_json::nlohmann_json)
target_compile_options(picker-library PRIVATE -Wall -Wextra -Wpedantic)
find_package(Threads REQUIRED)
add_executable(random-game-picker src/main.cpp src/http.cpp)
target_link_libraries(random-game-picker PRIVATE picker-library PkgConfig::SDL Threads::Threads)
target_compile_options(random-game-picker PRIVATE -Wall -Wextra -Wpedantic)
find_file(PICKER_FONT DejaVuSans.ttf PATHS /usr/share/fonts/truetype/dejavu REQUIRED)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/assets")
configure_file("${PICKER_FONT}" "${CMAKE_BINARY_DIR}/assets/DejaVuSans.ttf" COPYONLY)
include(CTest)
if(BUILD_TESTING)
  add_executable(library-test tests/library-test.cpp)
  target_link_libraries(library-test PRIVATE picker-library)
  add_test(NAME library-contract COMMAND library-test)
  add_executable(http-client-test tests/http-client.cpp src/http.cpp)
  target_include_directories(http-client-test PRIVATE src)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME loopback-contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/http-contract.py $<TARGET_FILE:http-client-test>)
endif()
